/* ============================ DECISION ENGINE ======================== */
const clamp = (x,a,b) => Math.max(a, Math.min(b, x));
const u = (x,w,b) => clamp((x-w)/(b-w), 0, 1);
const erf = x => { const s = x<0?-1:1; x = Math.abs(x);
  const t = 1/(1+0.3275911*x), y = 1-(((((1.061405429*t-1.453152027)*t)+1.421413741)*t-0.284496736)*t+0.254829592)*t*Math.exp(-x*x);
  return s*y; };
const ncdf = z => 0.5*(1+erf(z/Math.SQRT2));

function powerMean(s01, w, p) {
  p = (p===undefined) ? M.p : p;
  let acc = 0; for (let i=0;i<s01.length;i++) acc += w[i]*Math.pow(Math.max(s01[i],1e-9), p);
  return Math.pow(acc, 1/p);
}
const s01of = r => CRIT.map(c => (r[c]||0)/100);
const overall = (r, w, p) => 100*powerMean(s01of(r), w||WV, p);

function gatesOf(r) {
  const out = [];
  const push = g => out.push(g);
  if (r.operating_margin_percent < -5) push({id:'HC1_LOSS_MAKING', sev:'BLOCK',
    label:'Operation loses money at the forecast production rate',
    detail:'Operating margin '+r.operating_margin_percent.toFixed(1)+'% at forecast volume'});
  if (r.break_even_safety_margin_percent < 0) push({id:'HC2_BELOW_BREAKEVEN', sev:'CONDITION',
    label:'Forecast production sits below the break-even volume',
    detail:'Break-even '+fmt(r.break_even_tonnes)+' t/month vs forecast '+fmt(r.forecast_production_tonnes_month)+' t/month'});
  if (r.resource_category === 'Inferred' && r.geological_confidence < 45) push({id:'HC3_RESERVE_CONFIDENCE', sev:'REVIEW',
    label:'Geological confidence too low to support a mining decision',
    detail:'Inferred resource, geological confidence '+Math.round(r.geological_confidence)+'/100'});
  if (r.dumpers_required > r.dumpers_available + r.standby_capacity + 2 || r.excavators_required > r.excavators_available + 1)
    push({id:'HC4_FLEET_INFEASIBLE', sev:'CONDITION', label:'Required fleet cannot be assembled from available plus standby units',
      detail:'Need '+r.dumpers_required+' dumpers / '+r.excavators_required+' excavators, have '+r.dumpers_available+' / '+r.excavators_available+' plus '+r.standby_capacity+' standby'});
  if (r.shortfall_probability > 0.70 && r.shortfall_percent > 15) push({id:'HC5_SHORTFALL_RISK', sev:'CONDITION',
    label:'Unacceptably high probability of missing the production target',
    detail:'P(shortfall) '+Math.round(100*r.shortfall_probability)+'% with expected shortfall '+r.shortfall_percent.toFixed(1)+'%'});
  if (r.data_confidence_score < 40) push({id:'HC6_DATA_FLOOR', sev:'REVIEW',
    label:'Model and data confidence below the decision floor',
    detail:'Data and model confidence '+Math.round(r.data_confidence_score)+'/100'});
  if (r.reserve_estimate_tonnes < 120000) push({id:'HC7_SUB_SCALE_RESERVE', sev:'BLOCK',
    label:'Reserve too small to justify a standalone block development',
    detail:'Estimated reserve '+Math.round(r.reserve_estimate_tonnes/1000)+' kt'});
  if (r.trafficability_wet_season < 25 && r.mining_method === 'OC') push({id:'HC8_WET_SEASON_ACCESS', sev:'CONDITION',
    label:'Wet-season ground conditions prevent reliable haulage',
    detail:'Wet-season trafficability '+Math.round(r.trafficability_wet_season)+'/100'});
  return out;
}

function confidenceOf(r, gs) {
  const fc = Math.max(r.forecast_production_tonnes_month, 1);
  const width = (r.production_upper_bound - r.production_lower_bound)/fc;
  const interval = Math.max(0, 1-(width-0.25)/0.95);
  const stale = (r.data_quality_flag === 'DEGRADED' || r.data_quality_flag === 'STALE') ? 1 : 0;
  let c = 0.58*(r.data_confidence_score/100) + 0.27*Math.min(1, interval) + 0.15*(1-0.5*stale);
  c -= 0.10*gs.filter(g => g.sev === 'REVIEW').length;
  return Math.round(1000*clamp(c,0,1))/10;
}

const RANK = {GO:0, GO_WITH_CONDITIONS:1, REVIEW_MORE_DATA:2, NO_GO:3};
const capV = (v,c) => RANK[v] >= RANK[c] ? v : c;

function verdictOf(score, conf, gs) {
  let v = score >= M.bands.go ? 'GO' : score >= M.bands.conditional ? 'GO_WITH_CONDITIONS'
        : score >= M.bands.review ? 'REVIEW_MORE_DATA' : 'NO_GO';
  const why = ['Band verdict from overall feasibility '+score.toFixed(1)+'/100'];
  gs.forEach(g => { const before = v;
    if (g.sev === 'BLOCK') { v = 'NO_GO'; if (v !== before) why.push(g.id+' forces no-go: '+g.label); }
    else if (g.sev === 'REVIEW') { if (v !== 'NO_GO') v = capV(v,'REVIEW_MORE_DATA');
      if (v !== before) why.push(g.id+' forces review / more data: '+g.label); }
    else { v = capV(v,'GO_WITH_CONDITIONS');
      if (v !== before) why.push(g.id+' caps the verdict at conditional: '+g.label); }
  });
  if (conf < M.confReview && (v === 'GO' || v === 'GO_WITH_CONDITIONS')) {
    v = 'REVIEW_MORE_DATA'; why.push('Decision confidence '+conf.toFixed(0)+'/100 is below the '+M.confReview+' floor for a positive verdict');
  } else if (conf < M.confGo && v === 'GO') {
    v = 'GO_WITH_CONDITIONS'; why.push('Decision confidence '+conf.toFixed(0)+'/100 downgrades go to conditional');
  }
  return {verdict:v, why:why};
}

function assess(r, w, p) {
  const gs = gatesOf(r), sc = overall(r, w, p), cf = confidenceOf(r, gs);
  const vv = verdictOf(sc, cf, gs);
  return {score:sc, conf:cf, verdict:vv.verdict, why:vv.why, gates:gs};
}

/* exact Shapley over 7 criteria: 128 coalitions, enumerated */
function shapley(r) {
  const n = CRIT.length, s = s01of(r), base = CRIT.map(c => M.baseline[c]);
  const cache = new Map();
  const val = mask => { if (cache.has(mask)) return cache.get(mask);
    const x = base.slice(); for (let i=0;i<n;i++) if (mask & (1<<i)) x[i] = s[i];
    const v = 100*powerMean(x, WV); cache.set(mask, v); return v; };
  const fact = [1,1,2,6,24,120,720,5040];
  const phi = new Array(n).fill(0);
  for (let mask=0; mask<(1<<n); mask++) {
    const k = (mask.toString(2).match(/1/g)||[]).length;
    for (let i=0;i<n;i++) if (!(mask & (1<<i))) {
      phi[i] += fact[k]*fact[n-k-1]/fact[n]*(val(mask|(1<<i)) - val(mask));
    }
  }
  const o = {}; CRIT.forEach((c,i) => o[c] = phi[i]); return o;
}

/* counterfactual transfer functions - ported from decision_engine.apply_driver */
function applyDriver(r0, driver, value) {
  const r = Object.assign({}, r0);
  const tgt = r.production_target_tonnes_month, fc0 = r0.forecast_production_tonnes_month;
  function reprod(newFc, newSigma) {
    const s = newSigma === undefined ? r.forecast_sigma : newSigma;
    r.forecast_production_tonnes_month = newFc;
    r.shortfall_tonnes = Math.max(0, tgt-newFc);
    r.shortfall_percent = 100*r.shortfall_tonnes/Math.max(tgt,1);
    r.shortfall_probability = ncdf((Math.log(Math.max(tgt,1))-Math.log(Math.max(newFc,1)))/Math.max(s,1e-6));
    r.forecast_sigma = s;
    r.production_lower_bound = newFc*Math.exp(-1.2816*s);
    r.production_upper_bound = newFc*Math.exp(1.2816*s);
    const ci = (r.production_upper_bound-r.production_lower_bound)/Math.max(newFc,1);
    r.production_reliability_score = 100*(0.50*(1-u(r.shortfall_probability,0.05,0.80))
      + 0.30*(1-u(r.shortfall_percent,2,35)) + 0.20*(1-u(ci,0.25,1.10)));
  }
  function reecon() {
    const price = r.ore_price_per_tonne, vc = r.variable_cost_per_tonne, fx = r.fixed_cost_month;
    const fc = r.forecast_production_tonnes_month, rev = fc*price, cost = fx+vc*fc;
    r.monthly_revenue = rev; r.monthly_operating_cost = cost; r.operating_margin = rev-cost;
    r.operating_margin_percent = 100*(rev-cost)/Math.max(rev,1);
    const contrib = price-vc;
    r.break_even_tonnes = contrib > 1 ? fx/contrib : null;
    r.break_even_price = fx/Math.max(fc,1)+vc;
    r.break_even_safety_margin_percent = (r.break_even_tonnes != null && isFinite(r.break_even_tonnes))
      ? Math.max(-300, 100*(fc-r.break_even_tonnes)/Math.max(fc,1)) : -300;
    r.break_even_price_margin_percent = 100*(price-r.break_even_price)/Math.max(price,1);
    r.economic_viability_score = 100*(0.40*u(r.operating_margin_percent,-30,55)
      + 0.34*u(r.break_even_safety_margin_percent,-45,75) + 0.26*u(r.break_even_price_margin_percent,-25,60));
  }
  if (driver === 'availability') {
    const a0 = r0.equipment_availability_percent/100, a1 = value/100;
    r.equipment_availability_percent = value;
    const s0 = 0.05+0.22*(1-a0), s1 = 0.05+0.22*(1-a1);
    const sig = Math.sqrt(Math.max(r0.forecast_sigma**2-s0**2, 1e-4)+s1**2);
    const fleetBinds = ['Dumper','Excavator','Crusher'].indexOf(r0.bottleneck_equipment) >= 0;
    reprod(fleetBinds ? fc0*(a1/Math.max(a0,1e-6)) : fc0, sig);
    r.equipment_adequacy_score = 100*(0.38*u(r.fleet_adequacy_score,35,96)+0.28*u(a1,0.58,0.92)
      + 0.20*u(r.equipment_utilization_percent/100,0.35,0.85)+0.14*u(r.standby_capacity,0,3));
    const dc = r0.maintenance_cost_per_tonne*0.9*(Math.max(0,0.80-a1)-Math.max(0,0.80-a0));
    r.cost_per_tonne = r0.cost_per_tonne+dc;
    r.variable_cost_per_tonne = r0.variable_cost_per_tonne*(r.cost_per_tonne/Math.max(r0.cost_per_tonne,1));
    reecon();
  } else if (driver === 'rainimpact') {
    const base = 16.5*(1-r0.trafficability_index/100);
    const newFc = fc0*(1-value/100)/Math.max(1-base/100, 1e-6);
    const ws = Math.min(1, value/16.5);
    r.trafficability_index = 100*(1-ws*0.70);
    const s0 = 0.03+0.24*(base/16.5), s1 = 0.03+0.24*ws;
    reprod(newFc, Math.sqrt(Math.max(r0.forecast_sigma**2-s0**2,1e-4)+s1**2));
    r.environmental_feasibility_score = Math.max(0, r0.environmental_feasibility_score-1.9*(value-base));
    r.operational_feasibility_score = Math.max(0, r0.operational_feasibility_score-1.1*(value-base));
    reecon();
  } else if (driver === 'price') {
    r.ore_price_per_tonne = value; reecon();
  } else if (driver === 'cost') {
    const k = value/Math.max(r0.cost_per_tonne,1);
    r.cost_per_tonne = value; r.variable_cost_per_tonne = r0.variable_cost_per_tonne*k;
    r.fixed_cost_month = r0.fixed_cost_month*k; reecon();
  } else if (driver === 'dumpers') {
    const n0 = Math.max(1, r0.dumpers_available), n1 = Math.max(1, Math.round(value));
    r.dumpers_available = n1;
    if (r0.bottleneck_equipment === 'Dumper') reprod(fc0*n1/n0);
    r.fleet_adequacy_score = Math.min(100, r0.fleet_adequacy_score*(0.75+0.25*n1/n0));
    r.equipment_adequacy_score = Math.min(100, 100*(0.38*u(r.fleet_adequacy_score,35,96)
      + 0.28*u(r.equipment_availability_percent/100,0.58,0.92)
      + 0.20*u(r.equipment_utilization_percent/100,0.35,0.85)+0.14*u(r.standby_capacity,0,3)));
    reecon();
  } else if (driver === 'geoconf') {
    const g0 = r0.geological_confidence/100, g1 = value/100;
    r.geological_confidence = value;
    r.reserve_attractiveness_score = Math.max(0, r0.reserve_attractiveness_score+12*(u(g1,0.20,0.90)-u(g0,0.20,0.90)));
    const s0 = 0.04+0.16*(1-g0), s1 = 0.04+0.16*(1-g1);
    const sig = Math.sqrt(Math.max(r0.forecast_sigma**2-s0**2,1e-4)+s1**2);
    reprod(fc0, sig);
    r.data_confidence_score = 100*(0.36*u(g1,0.25,0.92)+0.24*u(r.geological_data_quality,25,92)
      + 0.22*(1-u(sig,0.12,0.46))+0.18*u(r.sampling_density||3,0.5,18));
  }
  return r;
}

function findFlip(r0, d) {
  const base = assess(r0).verdict, start = d.get(r0);
  const ends = [d.min, d.max];
  let best = null;
  ends.forEach(end => {
    let prev = start, hit = null;
    for (let i=1;i<=24;i++) {
      const x = start+(end-start)*i/24;
      if (assess(applyDriver(r0, d.k, x)).verdict !== base) { hit = [prev, x]; break; }
      prev = x;
    }
    if (hit) {
      let lo = hit[0], hi = hit[1];
      for (let i=0;i<16;i++) { const mid = (lo+hi)/2;
        if (assess(applyDriver(r0, d.k, mid)).verdict !== base) hi = mid; else lo = mid; }
      const nv = assess(applyDriver(r0, d.k, hi)).verdict;
      const gap = Math.abs(hi-start)/Math.max(Math.abs(start),1e-6);
      if (!best || gap < best.gap) best = {at:hi, to:nv, gap:gap, dir:hi<start?'falls below':'exceeds'};
    }
  });
  return best;
}

  const w = 420, h = 26+parts.length*15; let s = svgOpen(w,h), x = 0;

  return s+'</svg>';

function passes(r) {
  const d = decisionFor(r.location_id);
  if (S.f.mine.size && !S.f.mine.has(r.mine_name)) return false;
  if (S.f.verdict.size && !S.f.verdict.has(d.verdict)) return false;
  if (S.f.method.size && !S.f.method.has(r.mining_method)) return false;
  if (r.reserve_estimate_tonnes/1000 < S.f.res) return false;
  if (r.grade_mn_percent < S.f.grade) return false;
  if (r.geological_confidence < S.f.conf) return false;
  if (r.environmental_severity_score > S.f.env) return false;
  if (100*r.shortfall_probability > S.f.sf) return false;
  if (r.economic_viability_score < S.f.ec) return false;
  if (S.q) { const t = (r.location_id+' '+r.mine_name+' '+r.block_id).toLowerCase();
    if (t.indexOf(S.q.toLowerCase()) < 0) return false; }
  return true;
}

function fnv(s){let h=2166136261;for(let i=0;i<s.length;i++){h^=s.charCodeAt(i);h=Math.imul(h,16777619);}return h>>>0;}
function rngOf(x){let a=(x>>>0)||1;return function(){a^=a<<13;a>>>=0;a^=a>>>17;a^=a<<5;a>>>=0;return a/4294967296;};}
const F1=v=>v.toFixed(1);

function spline(p, closed){
  const n=p.length, P=i=>p[((i%n)+n)%n];
  let d='M'+F1(P(0)[0])+' '+F1(P(0)[1]);
  const last = closed ? n : n-1;
  for(let i=0;i<last;i++){
    const a=P(i-1),b=P(i),c=P(i+1),e=P(i+2);
    d+='C'+F1(b[0]+(c[0]-a[0])/6)+' '+F1(b[1]+(c[1]-a[1])/6)+' '
       +F1(c[0]-(e[0]-b[0])/6)+' '+F1(c[1]-(e[1]-b[1])/6)+' '+F1(c[0])+' '+F1(c[1]);
  }
  return d+(closed?'Z':'');
}
function ring(cx,cy,rx,ry,n,rough,rnd,ph){
  const p=[];
  for(let i=0;i<n;i++){const a=(ph||0)+2*Math.PI*i/n, k=1+(rough||0)*(rnd()*2-1);
    p.push([cx+rx*k*Math.cos(a), cy+ry*k*Math.sin(a)]);}
  return p;
}
function nestedRings(cx,cy,rx,ry,levels,shrink,rnd,rough){
  const n=18, noise=[]; for(let i=0;i<n;i++) noise.push(1+(rough||0.15)*(rnd()*2-1));
  const out=[];
  for(let L=0;L<levels;L++){
    const f=1-L*shrink, p=[];
    for(let i=0;i<n;i++){const a=2*Math.PI*i/n;
      p.push([cx+rx*f*noise[i]*Math.cos(a), cy+ry*f*noise[i]*Math.sin(a)]);}
    out.push(p);
  }
  return out;
}
/* short tick marks on one side of a path - bench faces / outcrop dip ticks */
function ticks(pts, len, inward, every){
  let s='';
  for(let i=0;i<pts.length;i+=(every||1)){
    const a=pts[i], b=pts[(i+1)%pts.length];
    const dx=b[0]-a[0], dy=b[1]-a[1], L=Math.hypot(dx,dy)||1;
    const nx=(inward? dy:-dy)/L, ny=(inward? -dx:dx)/L;
    s+='<line x1="'+F1(a[0])+'" y1="'+F1(a[1])+'" x2="'+F1(a[0]+nx*len)+'" y2="'+F1(a[1]+ny*len)+'"/>';
  }
  return s;
}
function leader(x1,y1,x2,y2,label,anchor){
  return '<g class="mm-ann"><path d="M'+F1(x1)+' '+F1(y1)+'L'+F1(x2)+' '+F1(y2)+'" marker-end="url(#mmArrow)"/>'
   +'<text x="'+F1(x1)+'" y="'+F1(y1-3)+'" text-anchor="'+(anchor||'middle')+'">'+esc(label)+'</text></g>';
}

function chips(host, values, key, labels) {
  el(host).innerHTML = values.map(v =>
    '<button class="chip" data-v="'+esc(v)+'" aria-pressed="'+S.f[key].has(v)+'">'+esc(labels?labels[v]:v)+'</button>').join('');
  el(host).querySelectorAll('.chip').forEach(b => b.addEventListener('click', () => {
    const v = b.dataset.v; S.f[key].has(v) ? S.f[key].delete(v) : S.f[key].add(v); render(); }));
}

function renderP1() {
  const vis = visible();
  const cnt = vis.length+' of '+LOCS.length+' shown';
  el('fcount').textContent = cnt; el('fcount2').textContent = cnt;
  chips('f-mine', M.mines, 'mine');
  chips('f-verdict', ['GO','GO_WITH_CONDITIONS','REVIEW_MORE_DATA','NO_GO'], 'verdict', VTXT);
  chips('f-method', ['OC','UG'], 'method', {OC:'Opencast', UG:'Underground'});

  // KPI strip
  const ds = vis.map(r => decisionFor(r.location_id));
  const c = v => ds.filter(d => d.verdict === v).length;
  const totRes = vis.reduce((a,r) => a+r.reserve_estimate_tonnes, 0);
  const totFc = vis.reduce((a,r) => a+r.forecast_production_tonnes_month, 0);
  const aggSf = vis.length ? vis.reduce((a,r) => a+r.shortfall_probability*r.production_target_tonnes_month, 0)
                             / Math.max(vis.reduce((a,r) => a+r.production_target_tonnes_month,0),1) : 0;
  const K = [['Potential reserve', fmt(totRes/1e6,2)+' Mt'], ['Locations assessed', vis.length],
    ['Go', c('GO')], ['Conditional', c('GO_WITH_CONDITIONS')], ['Review', c('REVIEW_MORE_DATA')],
    ['No-go', c('NO_GO')], ['Tonnage-weighted shortfall risk', (100*aggSf).toFixed(0)+'%']];
  el('kpis').innerHTML = K.map(k => '<div class="kpi"><div class="l">'+k[0]+'</div><div class="v">'+k[1]+'</div></div>').join('')
    + '<div class="kpi" style="grid-column:1/-1"><div class="l">Portfolio forecast at the selected scenario</div><div class="v">'
    + fmt(totFc)+' t/month <span style="font-size:11px;color:var(--steel);font-family:var(--sans)">across the filtered blocks</span></div></div>';

  // table
  mmHide();
  const tb = el('loctable').querySelector('tbody');
  tb.innerHTML = vis.sort((a,b) => decisionFor(b.location_id).score-decisionFor(a.location_id).score).map(r => {
    const d = decisionFor(r.location_id);
    return '<tr data-loc="'+r.location_id+'" data-sel="'+(r.location_id === S.sel ? 1 : 0)+'">'
      + '<td class="mono">'+r.location_id+'<span style="color:var(--mute)"> '+r.block_id+'</span></td>'
      + '<td>'+r.mine_name+'</td><td>'+r.mining_method+'</td>'
      + '<td class="n">'+fmt(r.reserve_estimate_tonnes/1000)+'</td>'
      + '<td class="n">'+r.grade_mn_percent.toFixed(1)+'</td>'
      + '<td class="n">'+Math.round(r.geological_confidence)+'</td>'
      + '<td class="n">'+Math.round(r.environmental_severity_score)+'</td>'
      + '<td class="n">'+Math.round(100*r.shortfall_probability)+'%</td>'
      + '<td class="n">'+Math.round(r.economic_viability_score)+'</td>'
      + '<td class="n"><b>'+d.score.toFixed(0)+'</b></td>'
      + '<td class="n">'+d.conf.toFixed(0)+'</td>'
      + '<td><span class="vpill" style="background:'+VCOL[d.verdict]+'">'+VTXT[d.verdict]+'</span></td>'
      + '<td class="mmcell"><button class="mmbtn" type="button" data-mm="'+r.location_id+'" '
      + 'aria-label="Show mine plan for '+r.location_id+'">'+minePlanThumb(r)+'<span>Plan</span></button></td>'
      + '<td style="color:var(--steel)">'+r.bottleneck_equipment.replace(/_/g,' ')+'</td></tr>';
  }).join('') || '<tr><td colspan="14" class="empty">No blocks match these filters. Widen a range or clear the filters.</td></tr>';
  tb.querySelectorAll('tr[data-loc]').forEach(tr => tr.addEventListener('click', ev => {
    if (ev.target.closest('.mmbtn')) return; S.sel = tr.dataset.loc; render(); }));

    b.addEventListener('mouseenter', ev => mmShow(ev, lid));
    b.addEventListener('focus',      ev => mmShow(ev, lid));
    b.addEventListener('mouseleave', mmHide);
    b.addEventListener('blur',       mmHide);
    b.addEventListener('click', ev => { ev.stopPropagation(); S.sel = lid; render(); });

  el('selpanel').innerHTML =

    + (r.mining_method === 'OC' ? 'Opencast' : 'Underground')+' · '+r.district+'</div>'
    + '<div class="verdictband" style="background:'+VCOL[d.verdict]+'"><div><div class="sc">'+d.score.toFixed(0)
    + '<span style="font-size:13px;opacity:.7">/100</span></div><div class="vt">'+VTXT[d.verdict]+'</div></div>'
    + '<div style="text-align:right;font-size:11px">confidence<br><b class="num" style="font-size:16px">'+d.conf.toFixed(0)+'</b></div></div>'
    + kv('Reserve', fmt(r.reserve_estimate_tonnes/1e6,2)+' Mt')
    + kv('Mn grade', r.grade_mn_percent.toFixed(1)+'%')
    + kv('Geological confidence', Math.round(r.geological_confidence)+'/100')

    + kv('Shortfall risk', Math.round(100*r.shortfall_probability)+'%')
    + kv('Economic viability', Math.round(r.economic_viability_score)+'/100')

    + '<button class="btn ghost" data-go="p3">Go to final verdict</button></div>'
    + '<p class="note" style="margin-top:12px">Hover the <b>Mine plan</b> cell in any row for that block’s schematic plan. '
    + 'Verdicts are computed under the <b>'+S.scen.replace(/_/g,' ')+'</b> scenario.</p>';
  el('selpanel').querySelectorAll('[data-go]').forEach(b => b.addEventListener('click', () => go(b.dataset.go)));

function renderP3() {
  locHeader('p3hdr');
  const r = cur(), d = assess(r), sh = shapley(r);

  el('bigcard').style.background = VCOL[d.verdict];
  el('bigcard').innerHTML = '<div><div style="font-size:11px;opacity:.85">Should MOIL mine this block?</div>'
    + '<div class="score">'+d.score.toFixed(0)+'<small>/100</small></div><div class="verd">'+VTXT[d.verdict]+'</div></div>'
    + '<div class="conf">Decision confidence '+d.conf.toFixed(0)+'/100<div class="confbar"><i style="width:'+d.conf+'%"></i></div>'
    + '<div style="margin-top:12px;font-size:11px;opacity:.9">'+esc(d.why[d.why.length-1])+'</div>'
    + '<div style="margin-top:10px;font-size:10px;opacity:.75">Fully compensatory weighted sum would score '
    + overall(r, WV, 1).toFixed(1)+'. The partially compensatory aggregation penalises weak dimensions.</div></div>';

  el('dims').innerHTML = CRIT.map(c => { const v = r[c], b = 100*M.baseline[c], s = sh[c];
    return '<div class="dimrow"><span>'+M.labels[c]+'</span>'
      + '<div class="bar"><i style="width:'+v+'%;background:'+(v >= 70 ? 'var(--go)' : v >= 50 ? 'var(--steel)' : 'var(--nogo)')+'"></i>'
      + '<u style="left:'+b+'%" title="portfolio median"></u></div>'
      + '<span class="sc">'+Math.round(v)+'</span>'
      + '<span class="sh '+(s >= 0 ? 'pos' : 'neg')+'">'+(s >= 0 ? '+' : '')+s.toFixed(2)+' pts</span></div>'; }).join('')
    + '<p class="note" style="margin-top:10px">Contributions are <b>exact Shapley values</b> over the seven criteria against the '
    + 'portfolio-median reference block. With seven criteria all 128 coalitions are enumerated, so there is no sampling error '
    + 'and no surrogate model. They sum to the gap between this block and the reference: '
    + CRIT.reduce((a,c) => a+sh[c], 0).toFixed(2)+' points.</p>';

  const pos = CRIT.map(c => [c, sh[c]]).filter(x => x[1] > 0.15).sort((a,b) => b[1]-a[1]).slice(0,3);
  const neg = CRIT.map(c => [c, sh[c]]).filter(x => x[1] < -0.15).sort((a,b) => a[1]-b[1]).slice(0,3);
  el('drivers').innerHTML = '<div style="font-size:11px;color:var(--go);font-weight:600;margin-bottom:4px">What supports the case</div>'
    + '<ul class="drv">'+(pos.map(p => '<li><span class="mk pos">+</span><span>'+M.labels[p[0]]+' at '+Math.round(r[p[0]])
      + '/100 <span class="num" style="color:var(--go)">'+p[1].toFixed(2)+' pts</span></span></li>').join('') || '<li><span>Nothing above the reference block.</span></li>')+'</ul>'
    + '<div style="font-size:11px;color:var(--nogo);font-weight:600;margin:10px 0 4px">What works against it</div>'
    + '<ul class="drv">'+(neg.map(p => '<li><span class="mk neg">\u2212</span><span>'+M.labels[p[0]]+' at '+Math.round(r[p[0]])
      + '/100 <span class="num" style="color:var(--nogo)">'+p[1].toFixed(2)+' pts</span></span></li>').join('') || '<li><span>Nothing below the reference block.</span></li>')+'</ul>';

  el('gates').innerHTML = d.gates.length
    ? d.gates.map(g => '<div class="gate '+(g.sev === 'CONDITION' ? 'cond' : g.sev === 'REVIEW' ? 'rev' : '')+'">'
        + '<b>'+esc(g.label)+'</b>'+esc(g.detail)+'<div class="basis">'+g.id+' \u00B7 severity '+g.sev+'</div></div>').join('')
    : '<p class="note">No hard constraint is violated. The verdict rests on the weighted assessment alone.</p>';

  const conds = d.gates.filter(g => g.sev === 'CONDITION');
  el('conds').innerHTML = (d.verdict === 'GO_WITH_CONDITIONS' || conds.length)
    ? '<p style="font-size:11.5px;color:var(--steel);margin:0 0 7px">A conditional go stands only once these are closed out:</p>'
      + '<ul class="drv">'+(conds.length ? conds.map(g => '<li><span class="mk">\u25A1</span><span>'+esc(g.label)+'</span></li>').join('')
        : '<li><span class="mk">\u25A1</span><span>Complete the recommended actions below before committing the plan.</span></li>')+'</ul>'
    : d.verdict === 'GO' ? '<p class="note">Unconditional go. Re-assess if any driver below moves past its threshold.</p>'
    : '<p class="note">The block is not currently in a conditional state; close the constraints above first.</p>';

  el('acts').innerHTML = actionsFor(r).map(a => '<li><b>'+esc(a.action)+'</b>'
    + '<span class="why">'+esc(a.why)+'</span><span class="eff">'+esc(a.eff)+'</span></li>').join('')
    || '<li>No mitigation is indicated at the current evidence.</li>';

  // counterfactual sliders
  el('cfs').innerHTML = DRIVERS.map(dr => {
    const start = dr.get(r), v = (S.cf[dr.k] !== undefined) ? S.cf[dr.k] : start;
    const flip = findFlip(r, dr);
    return '<div class="cf"><div class="hd"><span>'+dr.label+'</span><b>'+dr.fmt(v)+'</b></div>'
      + '<input type="range" data-d="'+dr.k+'" min="'+dr.min+'" max="'+dr.max+'" step="'+dr.step+'" value="'+v+'">'
      + '<div class="out"><span>now '+dr.fmt(start)+'</span>'
      + (flip ? '<span class="flip">verdict changes to '+VTXT[flip.to]+' when it '+flip.dir+' '+dr.fmt(flip.at)+'</span>'
              : '<span>no verdict change within the tested range</span>')+'</div></div>'; }).join('')
    + '<button class="resetbtn" id="cfreset" style="margin:10px 0 0">Reset drivers</button>';

  el('cfs').querySelectorAll('input[type=range]').forEach(inp => inp.addEventListener('input', () => {
    S.cf[inp.dataset.d] = parseFloat(inp.value); renderCfOut(); 
    inp.closest('.cf').querySelector('.hd b').textContent = DRIVERS.find(x => x.k === inp.dataset.d).fmt(parseFloat(inp.value)); }));
  el('cfreset').addEventListener('click', () => { S.cf = {}; renderP3(); });
  renderCfOut();

  // weight sensitivity
  const alts = [['AHP weights (default)', WV],
    ['Equal weights', CRIT.map(() => 1/CRIT.length)],
    ['Economics-led (+50% on economics)', normw(WV.map((w,i) => CRIT[i] === 'economic_viability_score' ? w*1.5 : w))],
    ['Supply-reliability-led (+50% on production reliability)', normw(WV.map((w,i) => CRIT[i] === 'production_reliability_score' ? w*1.5 : w))],
    ['Risk-averse (+50% on environment and data confidence)', normw(WV.map((w,i) => ['environmental_feasibility_score','data_confidence_score'].indexOf(CRIT[i]) >= 0 ? w*1.5 : w))],
    ['Fully compensatory (weighted sum, p = 1)', WV]];
  el('wsens').innerHTML = '<table><thead><tr><th>Weighting</th><th style="text-align:right">Score</th><th>Verdict</th>'
    + '<th style="text-align:right">Portfolio go count</th><th style="text-align:right">Rank of this block</th></tr></thead><tbody>'
    + alts.map((a,i) => { const p = (i === 5) ? 1 : M.p;
        const sc = overall(r, a[1], p), gs = gatesOf(r), cf = confidenceOf(r, gs), vv = verdictOf(sc, cf, gs).verdict;
        const all = LOCS.map(l => ({l:l, s:overall(rec(l), a[1], p)})).sort((x,y) => y.s-x.s);
        const nGo = LOCS.filter(l => { const rr = rec(l), g2 = gatesOf(rr);
          return verdictOf(overall(rr, a[1], p), confidenceOf(rr, g2), g2).verdict === 'GO'; }).length;
        const rank = all.findIndex(x => x.l === S.sel)+1;
        return '<tr'+(i === 0 ? ' data-sel="1"' : '')+'><td>'+a[0]+'</td><td class="n">'+sc.toFixed(1)+'</td>'
          + '<td><span class="vpill" style="background:'+VCOL[vv]+'">'+VTXT[vv]+'</span></td>'
          + '<td class="n">'+nGo+'</td><td class="n">'+rank+' of '+LOCS.length+'</td></tr>'; }).join('')
    + '</tbody></table><p class="note" style="margin-top:9px">If the verdict is identical across every row, it is robust to the '
    + 'weighting. Where it is not, the verdict depends on a judgement about what matters and should be escalated rather than automated.</p>';
}
const normw = a => { const s = a.reduce((x,y) => x+y,0); return a.map(x => x/s); };

function renderCfOut() {
  const r0 = cur(); let r = r0;
  DRIVERS.forEach(d => { if (S.cf[d.k] !== undefined && Math.abs(S.cf[d.k]-d.get(r0)) > 1e-9) r = applyDriver(r, d.k, S.cf[d.k]); });
  const b = assess(r0), d = assess(r), changed = d.verdict !== b.verdict;
  el('cfout').innerHTML =
    '<div class="verdictband" style="background:'+VCOL[d.verdict]+'"><div><div class="sc">'+d.score.toFixed(0)
    + '<span style="font-size:13px;opacity:.7">/100</span></div><div class="vt">'+VTXT[d.verdict]+'</div></div>'
    + '<div style="text-align:right;font-size:11px">confidence<br><b class="num" style="font-size:16px">'+d.conf.toFixed(0)+'</b></div></div>'
    + (changed ? '<p class="note" style="border-left:3px solid var(--nogo);background:#FBF2F1">The verdict has changed from <b>'
        + VTXT[b.verdict]+'</b> to <b>'+VTXT[d.verdict]+'</b>.</p>'
      : '<p class="note">Verdict unchanged at <b>'+VTXT[b.verdict]+'</b>. Score moved '
        + (d.score-b.score >= 0 ? '+' : '')+(d.score-b.score).toFixed(1)+' points.</p>')
    + '<div style="margin-top:10px">'
    + kv('Forecast production', fmt(r.forecast_production_tonnes_month)+' t/mo  ('+(r.forecast_production_tonnes_month-r0.forecast_production_tonnes_month >= 0 ? '+' : '')+fmt(r.forecast_production_tonnes_month-r0.forecast_production_tonnes_month)+')')
    + kv('P(shortfall)', Math.round(100*r.shortfall_probability)+'%  ('+((r.shortfall_probability-r0.shortfall_probability) >= 0 ? '+' : '')+Math.round(100*(r.shortfall_probability-r0.shortfall_probability))+'pp)')
    + kv('Operating margin', r.operating_margin_percent.toFixed(1)+'%  ('+((r.operating_margin_percent-r0.operating_margin_percent) >= 0 ? '+' : '')+(r.operating_margin_percent-r0.operating_margin_percent).toFixed(1)+'pp)')
    + kv('Break-even safety margin', r.break_even_safety_margin_percent.toFixed(1)+'%')
    + kv('Economic viability', Math.round(r.economic_viability_score)+'/100')
    + kv('Production reliability', Math.round(r.production_reliability_score)+'/100')
    + '</div>'
    + '<div style="margin-top:10px;font-size:11px;color:var(--steel)"><b>Recommended actions under these conditions</b><ol style="margin:5px 0 0;padding-left:16px">'
    + (actionsFor(r).slice(0,3).map(a => '<li style="margin-bottom:3px">'+esc(a.action)+'</li>').join('') || '<li>None indicated.</li>')+'</ol></div>';
}

/* rule-based action engine, ported from decision_engine.ACTION_RULES */
function actionsFor(r) {
  const A = [];
  if (r.drainage_risk > 55 && r.wet_season_flag === 1) A.push({
    action:'Prepare haul-road drainage and wet-season surfacing before June',
    why:'Field studies record wet-road cycle-time increases of up to 15.6% and truck productivity losses of up to 13.1%.',
    eff:'about '+Math.min(9, 0.35*(100-r.haul_road_condition)*0.25).toFixed(1)+'% wet-season production recovered'});
  if ((r.bottleneck_equipment === 'Dumper' && r.standby_capacity < 2) || r.shortfall_probability > 0.55) A.push({
    action:'Add standby hauling capacity for the monsoon window',
    why:'Lifts the binding haulage constraint directly and cuts shortfall probability.',
    eff:'+'+Math.round(100/Math.max(r.dumpers_available,1))+'% haulage capacity per additional dumper'});
  if (r.seasonal_operating_risk > 55) A.push({
    action:'Rebalance the monthly target: reduce June to September, recover October to May',
    why:'Removes the mismatch between a flat target and a seasonal capacity curve, which is the dominant source of reported shortfall.',
    eff:'removes about '+Math.min(r.shortfall_percent,12).toFixed(1)+' percentage points of shortfall with no capex'});
  if (r.data_confidence_score < 60 || r.resource_category === 'Inferred') A.push({
    action:'Commission infill drilling to upgrade the resource category',
    why:'The only route from an Inferred resource to a reportable reserve, and it narrows the production interval.',
    eff:'target 12+ holes/km\u00B2 (now '+(r.sampling_density ? r.sampling_density.toFixed(1) : 'not recorded')+')'});
  if (r.blasting_delay_probability > 0.35 || (r.average_blasting_delay_hours || 0) > 3) A.push({
    action:'Re-sequence blasting to compress the delay window',
    why:'Recovers effective operating hours lost to charging, clearance and re-entry.',
    eff:'about '+Math.round(r.blasting_frequency*(r.average_blasting_delay_hours||0)*r.blasting_delay_probability*0.4)+' operating hours a month'});
  if (r.fleet_adequacy_score < 60) A.push({
    action:'Rematch the loader and truck fleet to remove the imbalance',
    why:'Loader-truck mismatch is a recognised primary source of queueing loss in shovel-truck systems.',
    eff:'bottleneck is '+r.bottleneck_equipment.replace(/_/g,' ')+'; need '+r.dumpers_required+' dumpers, have '+r.dumpers_available});
  if (r.break_even_safety_margin_percent < 25) A.push({
    action:'Re-run economics under a downside price and fuel case before committing capital',
    why:'A thin break-even margin means the verdict is price-fragile; the downside case should be the decision case.',
    eff:'break-even price \u20B9'+fmt(r.break_even_price)+'/t against realised \u20B9'+fmt(r.ore_price_per_tonne)+'/t'});
  if (r.shortfall_probability > 0.65 && r.seasonal_operating_risk > 50) A.push({
    action:'Shift committed tonnes to a higher-reliability block for the monsoon quarter',
    why:'Protects customer supply continuity, the stated business objective, without waiting for capex.',
    eff:'about '+fmt(r.shortfall_tonnes)+' t a month to source elsewhere'});
  if (r.equipment_availability_percent < 72) A.push({
    action:'Move the bottleneck machine onto condition-based maintenance',
    why:'Availability below the 78% peer-reviewed benchmark points to reactive rather than planned maintenance.',
    eff:'closing the gap to 78% is worth about '+Math.round(Math.max(0,78-r.equipment_availability_percent)/Math.max(r.equipment_availability_percent,1)*100)+'% capacity where the fleet binds'});
  return A.slice(0,5);
}
